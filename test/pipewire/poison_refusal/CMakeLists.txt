# Run by the ear_raw_pipewire_* tests (test/pipewire/CMakeLists.txt): a stand-in for PipeWire's imported target and
# targets that see its headers as EAR_SEES says. The refusal is asked for before any target is made, as src/pipewire
# asks before later directories are read, so every refusal here is of a target made after it. Nothing is built:
# configure stops at a refusal, and only generates otherwise.
cmake_minimum_required(VERSION 3.28...4.2)
project(PoisonRefusal LANGUAGES CXX)
include("${EAR_CMAKE_DIR}/RawCallbacks.cmake")

add_library(Stand::PIPEWIRE INTERFACE IMPORTED)
set(headers "${CMAKE_CURRENT_SOURCE_DIR}/headers")
set_target_properties(Stand::PIPEWIRE PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${headers}")
set(header "${CMAKE_CURRENT_SOURCE_DIR}/no_raw.hpp")
ear_refuse_raw_pipewire(Stand::PIPEWIRE "${header}" EXEMPT ear_listed_tests)

set(source "${CMAKE_CURRENT_SOURCE_DIR}/empty.cpp")
add_library(ear_helper INTERFACE)
add_library(ear_alias ALIAS ear_helper)
add_library(ear_user STATIC "${source}")
if(EAR_SEES STREQUAL "direct")
	target_link_libraries(ear_user PRIVATE Stand::PIPEWIRE)
elseif(EAR_SEES STREQUAL "executable")
	add_executable(ear_program "${source}")
	target_link_libraries(ear_program PRIVATE Stand::PIPEWIRE)
elseif(EAR_SEES STREQUAL "helper")
	target_link_libraries(ear_user PRIVATE ear_helper)
	target_link_libraries(ear_helper INTERFACE Stand::PIPEWIRE)
elseif(EAR_SEES STREQUAL "alias")
	target_link_libraries(ear_user PRIVATE ear_alias)
	target_link_libraries(ear_helper INTERFACE Stand::PIPEWIRE)
elseif(EAR_SEES STREQUAL "generator-expression")
	target_link_libraries(ear_user PRIVATE $<BUILD_INTERFACE:Stand::PIPEWIRE>)
elseif(EAR_SEES STREQUAL "include-directories")
	target_include_directories(ear_user PRIVATE "${headers}")
elseif(EAR_SEES STREQUAL "poisoned")
	target_link_libraries(ear_user PRIVATE Stand::PIPEWIRE)
	ear_poison_raw_callbacks(ear_user "${header}")
elseif(EAR_SEES STREQUAL "listed")
	add_executable(ear_listed_tests "${source}")
	target_link_libraries(ear_listed_tests PRIVATE Stand::PIPEWIRE)
elseif(EAR_SEES STREQUAL "private-link")
	# A program linking a poisoned static library that links PipeWire privately: the program sees no header.
	target_link_libraries(ear_user PRIVATE Stand::PIPEWIRE)
	ear_poison_raw_callbacks(ear_user "${header}")
	add_executable(ear_program "${source}")
	target_link_libraries(ear_program PRIVATE ear_user)
elseif(NOT EAR_SEES STREQUAL "none")
	message(FATAL_ERROR "Unknown EAR_SEES: ${EAR_SEES}")
endif()
